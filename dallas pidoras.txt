-- Получение активного оружия
local function getActiveTools()
    local tools = {}
    local backpack = localPlayer:FindFirstChild("Backpack")
    local char = localPlayer.Character
    
    local function checkAndAdd(name)
        local t = (backpack and backpack:FindFirstChild(name)) or (char and char:FindFirstChild(name))
        if t then table.insert(tools, t) end
    end

    if selectedWeaponMode == "Spas 12" then
        checkAndAdd("Spas 12")
    elseif selectedWeaponMode == "Remington" then
        checkAndAdd("Remington")
    elseif selectedWeaponMode == "Both" then
        checkAndAdd("Spas 12")
        checkAndAdd("Remington")
    end
    return tools
end

-- Получение любого доступного оружия
local function getTool()
    local tools = getActiveTools()
    return tools[1]
end

-- Временное отключение столкновений
local function setCharacterCollision(char, canCollide)
    if not char then return end
    for _, part in ipairs(char:GetDescendants()) do
        if part:IsA("BasePart") then
            part.CanCollide = canCollide
        end
    end
end

-- Автоматический подбор оружия
local function tryAutoEquipWeapon()
    local currentTools = getActiveTools()
   
    local needsSpas = false
    local needsRem = false
    
    if selectedWeaponMode == "Spas 12" then
        if #currentTools >= 1 then return end
        needsSpas = true
    elseif selectedWeaponMode == "Remington" then
        if #currentTools >= 1 then return end
        needsRem = true
    elseif selectedWeaponMode == "Both" then
        if #currentTools >= 2 then return end
        local hasSpas, hasRem = false, false
        for _, t in ipairs(currentTools) do
            if t.Name == "Spas 12" then hasSpas = true end
            if t.Name == "Remington" then hasRem = true end
        end
        needsSpas = not hasSpas
        needsRem = not hasRem
    end
    
    local char = localPlayer.Character
    if not char or not char:FindFirstChild("HumanoidRootPart") then return end
    local rootPart = char.HumanoidRootPart
    local originalCFrame = rootPart.CFrame
    
    local targetNames = {}
    if needsSpas then table.insert(targetNames, "spas 12") end
    if needsRem then table.insert(targetNames, "remington") end

    local foundTarget = nil
    for _, obj in pairs(Workspace:GetDescendants()) do
        local nameLower = string.lower(obj.Name)
        for _, tName in ipairs(targetNames) do
            if nameLower == tName then
                foundTarget = obj
                break
            end
        end
        if foundTarget then break end
    end

    if foundTarget then
        local targetCFrame = nil
        if foundTarget:IsA("Model") and foundTarget.PrimaryPart then
            targetCFrame = foundTarget.PrimaryPart.CFrame
        elseif foundTarget:IsA("BasePart") then
            targetCFrame = foundTarget.CFrame
        elseif foundTarget:IsA("Tool") and foundTarget:FindFirstChild("Handle") then
            targetCFrame = foundTarget.Handle.CFrame
        else
            local part = foundTarget:FindFirstChildWhichIsA("BasePart", true)
            if part then targetCFrame = part.CFrame end
        end

        if targetCFrame then
            setCharacterCollision(char, false)
            rootPart.CFrame = targetCFrame + Vector3.new(0, 1, 0)
            task.wait(0.2)
        
            local prompt = foundTarget:FindFirstChildWhichIsA("ProximityPrompt", true)
            if prompt and fireproximityprompt then fireproximityprompt(prompt) end
            local clickDetector = foundTarget:FindFirstChildWhichIsA("ClickDetector", true)
            if clickDetector and fireclickdetector then fireclickdetector(clickDetector) end
            task.wait(0.2)
            rootPart.CFrame = originalCFrame
            setCharacterCollision(char, true)
        end
    end
end

-- Управление движущейся платформой
local function updateMovingPlatform()
    if movingPlatformEnabled then
        local char = localPlayer.Character
        local humanoid = char and char:FindFirstChildOfClass("Humanoid")
        if not humanoid or humanoid.Health <= 0 then
            if movingPlatform then movingPlatform:Destroy(); movingPlatform = nil end
            originalPosition = nil
            return
        end

        if not movingPlatform or not movingPlatform.Parent then
            movingPlatform = Instance.new("Part")
            movingPlatform.Name = "АвтоПлатформа"
            movingPlatform.Size = Vector3.new(30, 2, 30)
            movingPlatform.Position = Vector3.new(10000, 5000, 10000)
            movingPlatform.Anchored = true
            movingPlatform.Material = Enum.Material.Neon
            movingPlatform.Color = Color3.fromRGB(0, 150, 255)
            movingPlatform.Parent = Workspace
        end
        local timeTick = os.clock() * (platformSpeed / 200) 
       
        movingPlatform.CFrame = CFrame.new(10000 + math.sin(timeTick * 0.05) * 500, 5000, 10000 + math.cos(timeTick * 0.05) * 500)
        
        if char and char:FindFirstChild("HumanoidRootPart") and not isHealing then
            local root = char.HumanoidRootPart
            if (root.Position - movingPlatform.Position).Magnitude > 40 then
                if not originalPosition then originalPosition = root.CFrame end
                setCharacterCollision(char, false)
                root.CFrame = movingPlatform.CFrame + Vector3.new(0, 4, 0)
            end
        end
    else
        if movingPlatform then
            movingPlatform:Destroy()
            movingPlatform = nil
            if originalPosition and not isHealing then
                local char = localPlayer.Character
                if char and char:FindFirstChild("HumanoidRootPart") then
                    char.HumanoidRootPart.CFrame = originalPosition
                end
                originalPosition = nil
            end
        end
    end
end

-- Перезарядка
local function reloadTool(tool)
    if not tool or isReloading then return end
    task.spawn(function()
        if ReloadEvent then
            pcall(function()
                ReloadEvent:InvokeServer(tool, "Start Reload")
                task.wait(reloadSpeed)
                ReloadEvent:InvokeServer(tool, "New Shell")
                task.wait(reloadSpeed)
                ReloadEvent:InvokeServer(tool, "Insert Shell")
                task.wait(reloadSpeed)
                ReloadEvent:InvokeServer(tool, "Stop Reload")
            end)
        end
    end)
end

-- Визуальные трассеры пуль
local function drawTracer(origin, target, weaponName)
    local distance = (target - origin).Magnitude
    local tracer = Instance.new("Part")
    tracer.Name = "Трассер"
    tracer.Anchored = true
    tracer.CanCollide = false
    tracer.CanQuery = false
    tracer.CastShadow = false
    tracer.Material = Enum.Material.Neon
    
    if weaponName == "Remington" then
        tracer.Color = Color3.fromRGB(50, 150, 255) -- Синий для Remington
    else
        tracer.Color = Color3.fromRGB(255, 180, 50) -- Оранжевый для Spas 12
    end
    
    tracer.Size = Vector3.new(0.08, 0.08, distance)
    tracer.CFrame = CFrame.lookAt(origin, target) * CFrame.new(0, 0, -distance / 2)
    tracer.Parent = Workspace
    
    local tween = TweenService:Create(tracer, TweenInfo.new(0.15), {
        Transparency = 1,
        Size = Vector3.new(0, 0, distance)
    })
    tween:Play()
    Debris:AddItem(tracer, 0.2)
end

local function isTargetAlive(player)
    if not player then return false end
    local char = player.Character
    if not char then return false end
    local head = char:FindFirstChild("Head")
    local humanoid = char:FindFirstChildOfClass("Humanoid")
    if not head or not humanoid then return false end
    if humanoid.Health <= 0 then return false end
    return true
end

-- Получение следующей цели
local function getNextExecutionTarget()
    if executionTarget and isTargetAlive(executionTarget) then
        local uid = executionTarget.UserId
        local isWhite = userStates[uid] and userStates[uid].state == "Белый"
        local isLock = userStates[uid] and userStates[uid].state == "Заблокирован"
        if not isWhite and (targetAllEnabled or isLock) then
            return executionTarget
        end
    end

    local candidates = {}
    for _, p in ipairs(Players:GetPlayers()) do
        if p == localPlayer then continue end
        local uid = p.UserId
        local isWhite = userStates[uid] and userStates[uid].state == "Белый"
        local isLock = userStates[uid] and userStates[uid].state == "Заблокирован"
        
        if isWhite or not isTargetAlive(p) then continue end
        if targetAllEnabled or isLock then
            table.insert(candidates, p)
        end
    end

    if #candidates == 0 then executionTarget = nil return nil end
    table.sort(candidates, function(a, b) return a.UserId < b.UserId end)

    local nextIdx = 1
    if executionTarget then
        for i, p in ipairs(candidates) do
            if p.UserId > executionTarget.UserId then nextIdx = i break end
        end
    end

    executionTarget = candidates[nextIdx]
    return executionTarget
end

-- Основная функция выстрела
local function fireGunAtTarget(tool, targetPlayer)
    local myChar = localPlayer.Character
    if not myChar or not myChar:FindFirstChild("HumanoidRootPart") then return end
    local muzzle = tool:FindFirstChild("Muzzle") or tool:FindFirstChild("Barrel") or tool:FindFirstChild("Handle")
    local origin = muzzle and muzzle.Position or myChar.HumanoidRootPart.Position

    local targetChar = targetPlayer.Character
    local head = targetChar and targetChar:FindFirstChild("Head")
    if not head then return end

    local headCenter = head.Position
    local mainDirection = (headCenter - origin).Unit
    local weaponName = tool.Name 

    if FireEvent then
        FireEvent:FireServer(tool, "Fire", {
            ["velocity"] = 2000,
            ["direction"] = mainDirection,
            ["bullet"] = game:GetService("ReplicatedStorage").Models.Bullets.Default,
            ["origin"] = origin
        })

        task.spawn(function()
            for i = 1, PELLET_COUNT do
                local offset = Vector3.new(
                    (math.random() - 0.5) * currentSpread,
                    (math.random() - 0.5) * currentSpread,
                    (math.random() - 0.5) * currentSpread
                )
                local aimPoint = headCenter + offset
                local direction = (aimPoint - origin).Unit
                
                drawTracer(origin, aimPoint, weaponName)

                if weaponName == "Remington" then
                    FireEvent:FireServer("Remington", "Hit", {
                        ["normal"] = -direction, 
                        ["bullet_cframe"] = CFrame.lookAt(aimPoint, aimPoint + direction), 
                        ["hit"] = head,
                        ["velocity"] = direction * 2000,
                        ["position"] = origin 
                    })
                else
                    FireEvent:FireServer(tool, "Hit", {
                        ["normal"] = -direction, 
                        ["bullet_cframe"] = CFrame.lookAt(aimPoint, aimPoint + direction), 
                        ["hit"] = head,
                        ["velocity"] = direction * 2000,
                        ["position"] = origin 
                    })
                end
            end
        end)
    end
end

-- Основная логика стрельбы
local function shoot()
    if isHealing then return end 
    local tools = getActiveTools()
    if #tools == 0 then return end

    local allValidTargets = {}
  
    if simultaneousShootEnabled then
        if targetAllEnabled then
            for _, p in ipairs(Players:GetPlayers()) do
                if p == localPlayer then continue end
                local uid = p.UserId
                local isWhite = userStates[uid] and userStates[uid].state == "Белый"
                if not isWhite and isTargetAlive(p) then
                    table.insert(allValidTargets, p)
                end
            end
        else
            for _, p in ipairs(Players:GetPlayers()) do
                if p == localPlayer then continue end
                local uid = p.UserId
                local isLock = userStates[uid] and userStates[uid].state == "Заблокирован"
                if isLock and isTargetAlive(p) then
                    table.insert(allValidTargets, p)
                end
            end
            
            if #allValidTargets == 0 then
                local t = getNextExecutionTarget()
                if t then table.insert(allValidTargets, t) end
            end
        end
    else
        local t = getNextExecutionTarget()
        if t then table.insert(allValidTargets, t) end
    end

    if #allValidTargets == 0 then return end

    if dualFireSyncMode == "Sequential" and #tools >= 2 then
        if nextGunIndex > #tools then nextGunIndex = 1 end
        local tool = tools[nextGunIndex]
        for _, target in ipairs(allValidTargets) do
            fireGunAtTarget(tool, target)
        end
        nextGunIndex = (nextGunIndex % #tools) + 1
    else
        if #tools == 2 and dualFocusMode == "Split" and #allValidTargets >= 2 then
            fireGunAtTarget(tools[1], allValidTargets[1])
            fireGunAtTarget(tools[2], allValidTargets[2])
        else
            for _, tool in ipairs(tools) do
                for _, target in ipairs(allValidTargets) do
                    fireGunAtTarget(tool, target)
                end
            end
        end
    end

    shotsFired = shotsFired + 1
end

-- Автоблок
local function blockLoop()
    while autoBlockEnabled do
        if SwingEvent then
            pcall(function() SwingEvent:FireServer("Block") end)
        end
        task.wait(0.1)
    end
end

-- Основной цикл мониторинга
task.spawn(function()
    while true do
        task.wait(0.1)
        local char = localPlayer.Character
        local humanoid = char and char:FindFirstChildOfClass("Humanoid")
        local rootPart = char and char:FindFirstChild("HumanoidRootPart")

        if not humanoid or humanoid.Health <= 0 then
            isHealing = false
            originalPosition = nil
            continue
        end

        if autoEquipEnabled then
            tryAutoEquipWeapon()
        end

        if autoHealEnabled and humanoid and rootPart then
            if humanoid.Health < humanoid.MaxHealth then
                if not isHealing then
                    isHealing = true
                    originalPosition = rootPart.CFrame
                end
                setCharacterCollision(char, false)
                rootPart.CFrame = CFrame.new(HEAL_POSITION)
            else
                if isHealing then
                    isHealing = false
                    setCharacterCollision(char, true)
                    if originalPosition and not movingPlatformEnabled then
                        rootPart.CFrame = originalPosition
                    end
                    originalPosition = nil
                end
            end
        else
            if isHealing then
                isHealing = false
                if char then setCharacterCollision(char, true) end
            end
        end

        updateMovingPlatform()

        if spectatingTarget and spectatingTarget.Character then
            local targetChar = spectatingTarget.Character
            local targetHumanoid = targetChar:FindFirstChildOfClass("Humanoid")
            if targetHumanoid then
                Camera.CameraSubject = targetHumanoid
            end
        end
    end
end)

local function autoLoop()
    while autoEnabled do
        if not isHealing then
            if shotsFired >= MAX_BEFORE_RELOAD then
                for _, tool in ipairs(getActiveTools()) do
                    reloadTool(tool)
                end
                shotsFired = 0
            end
            shoot()
        end
        task.wait(FIRE_RATE)
    end
end

local function getButtonColorForState(state)
    if state == "Заблокирован" then return Color3.fromRGB(180, 40, 40) end
    if state == "Белый" then return Color3.fromRGB(40, 150, 40) end
    return Color3.fromRGB(55, 55, 55)
end

-- Обновление списка игроков
local function refreshPlayerList()
    for _, child in ipairs(contentScroll:GetChildren()) do
        if child.Name == "PlayerRow" then child:Destroy() end
    end

    local allIds = {}
    for _, p in ipairs(Players:GetPlayers()) do
        if p ~= localPlayer then allIds[p.UserId] = { isOnline = true, p = p } end
    end
    for uid, info in pairs(userStates) do
        if not allIds[uid] then allIds[uid] = { isOnline = false, info = info } end
    end

    local order = 15
    for uid, data in pairs(allIds) do
        local state = userStates[uid] and userStates[uid].state or "Нет"
        local displayName = data.isOnline and data.p.DisplayName or data.info.displayName
        local name = data.isOnline and data.p.Name or data.info.name
        local prefix = data.isOnline and "" or "[ОФФЛАЙН] "

        local row = Instance.new("Frame")
        row.Name = "PlayerRow"
        row.Size = UDim2.new(1, -6, 0, 22)
        row.BackgroundTransparency = 1
        row.LayoutOrder = order
        row.Parent = contentScroll

        local toggleStateBtn = Instance.new("TextButton")
        toggleStateBtn.Size = UDim2.new(0.7, -2, 1, 0)
        toggleStateBtn.BackgroundColor3 = getButtonColorForState(state)
        toggleStateBtn.Text = prefix .. displayName
        toggleStateBtn.TextColor3 = data.isOnline and Color3.fromRGB(255, 255, 255) or Color3.fromRGB(150, 150, 150)
        toggleStateBtn.Font = Enum.Font.SourceSans
        toggleStateBtn.TextSize = 10
        toggleStateBtn.Parent = row

        toggleStateBtn.MouseButton1Click:Connect(function()
            if state == "Нет" then
                userStates[uid] = { state = "Заблокирован", displayName = displayName, name = name }
            elseif state == "Заблокирован" then
                userStates[uid].state = "Белый"
            elseif state == "Белый" then
                userStates[uid] = nil
            end
            if executionTarget and executionTarget.UserId == uid then
                executionTarget = nil
            end
            refreshPlayerList()
        end)

        local spectateBtn = Instance.new("TextButton")
        spectateBtn.Size = UDim2.new(0.3, -2, 1, 0)
        spectateBtn.Position = UDim2.new(0.7, 2, 0, 0)
        spectateBtn.BackgroundColor3 = Color3.fromRGB(50, 100, 150)
        spectateBtn.Text = "👁️"
        spectateBtn.TextColor3 = Color3.fromRGB(255, 255, 255)
        spectateBtn.Font = Enum.Font.SourceSansBold
        spectateBtn.TextSize = 10
        spectateBtn.Parent = row

        spectateBtn.MouseButton1Click:Connect(function()
            if data.isOnline and data.p then
                spectatingTarget = data.p
                spectateText.Text = "👁️ Наблюдение: " .. displayName
            end
        end)

        order = order + 1
    end
end

toggleBtn.MouseButton1Click:Connect(function()
    autoEnabled = not autoEnabled
    if autoEnabled then
        if autoThread then task.cancel(autoThread) end
        autoThread = task.spawn(autoLoop)
    else
        executionTarget = nil
        if autoThread then task.cancel(autoThread); autoThread = nil end
    end
    updateTopUI()
end)

    end
    updateTopUI()
end)
